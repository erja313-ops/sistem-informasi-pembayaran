<script>

document.addEventListener('DOMContentLoaded', function () {

    const sidebarToggle =
        document.getElementById('sidebarToggle');

    const body =
        document.body;


    /*
    =============================
    Ambil status terakhir
    =============================
    */

    const sidebarState =
        localStorage.getItem('adminSidebarCollapsed');


    if (sidebarState === 'true') {

        body.classList.add(
            'sidebar-collapsed'
        );

    }


    /*
    =============================
    Tombol buka / tutup
    =============================
    */

    sidebarToggle.addEventListener(
        'click',
        function () {

            body.classList.toggle(
                'sidebar-collapsed'
            );


            localStorage.setItem(
                'adminSidebarCollapsed',

                body.classList.contains(
                    'sidebar-collapsed'
                )
            );

        }
    );

});

</script>

<script>
document.addEventListener('DOMContentLoaded', function () {
    var root   = document.documentElement,
        body   = document.body,
        CK     = 'adminSidebarCollapsed',
        TK     = 'alAsrorTheme';

    var toggle  = document.getElementById('axToggle'),
        burger  = document.getElementById('axBurger'),
        closeBtn= document.getElementById('axClose'),
        backdrop= document.getElementById('axBackdrop'),
        sidebar = document.getElementById('axSidebar'),
        themeBtn= document.getElementById('axTheme');

    var mqMobile = window.matchMedia('(max-width:767.98px)');

    /* ---------- COLLAPSE (desktop) ---------- */
    if (localStorage.getItem(CK) === 'true') body.classList.add('ax-collapsed');
    function syncToggleAria(){ if(toggle) toggle.setAttribute('aria-expanded', String(!body.classList.contains('ax-collapsed'))); }
    syncToggleAria();
    if (toggle) toggle.addEventListener('click', function(){
        body.classList.toggle('ax-collapsed');
        localStorage.setItem(CK, body.classList.contains('ax-collapsed'));
        syncToggleAria();
    });

    /* ---------- INERT ---------- */
    function syncInert(){ if(sidebar) sidebar.inert = mqMobile.matches && !body.classList.contains('ax-drawer-open'); }

    /* ---------- DRAWER (mobile) ---------- */
    function openDrawer(){
        body.classList.add('ax-drawer-open');
        if(burger) burger.setAttribute('aria-expanded','true');
        syncInert();
        var f = sidebar && sidebar.querySelector('.ax-link');
        if(f) f.focus({preventScroll:true});
    }
    function closeDrawer(){
        body.classList.remove('ax-drawer-open');
        if(burger) burger.setAttribute('aria-expanded','false');
        syncInert();
        if(burger) burger.focus({preventScroll:true});
    }
    if(burger)  burger.addEventListener('click', openDrawer);
    if(closeBtn)closeBtn.addEventListener('click', closeDrawer);
    if(backdrop)backdrop.addEventListener('click', closeDrawer);
    document.addEventListener('keydown', function(e){
        if(e.key === 'Escape' && body.classList.contains('ax-drawer-open')) closeDrawer();
    });
    (mqMobile.addEventListener ? mqMobile.addEventListener.bind(mqMobile,'change') : mqMobile.addListener.bind(mqMobile))(function(e){
        var matches = (e.matches !== undefined) ? e.matches : mqMobile.matches;
        if(!matches) closeDrawer();
        syncInert();
    });
    document.addEventListener('keydown', function(e){
        if(e.key !== 'Tab' || !body.classList.contains('ax-drawer-open') || !sidebar) return;
        var els = sidebar.querySelectorAll('a[href],button:not([disabled])');
        if(!els.length) return;
        var first = els[0], last = els[els.length-1];
        if(e.shiftKey && document.activeElement === first){ e.preventDefault(); last.focus(); }
        else if(!e.shiftKey && document.activeElement === last){ e.preventDefault(); first.focus(); }
    });
    if(sidebar) sidebar.addEventListener('click', function(e){
        if(e.target.closest('.ax-link') && mqMobile.matches) closeDrawer();
    });
    syncInert();

    /* ---------- TEMA ---------- */
    function applyTheme(t){
        root.setAttribute('data-theme', t);
        localStorage.setItem(TK, t);
        if(themeBtn){
            var dark = (t === 'dark');
            themeBtn.querySelector('i').className = 'bi ' + (dark ? 'bi-sun-fill' : 'bi-moon-stars-fill');
            themeBtn.querySelector('span').textContent = dark ? 'Terang' : 'Gelap';
            themeBtn.setAttribute('aria-label', dark ? 'Aktifkan mode terang' : 'Aktifkan mode gelap');
        }
    }
    applyTheme(root.getAttribute('data-theme') || 'light');
    if(themeBtn) themeBtn.addEventListener('click', function(){
        applyTheme(root.getAttribute('data-theme') === 'dark' ? 'light' : 'dark');
    });
});
